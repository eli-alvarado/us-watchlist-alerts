        closes = [c for c in close if c is not None]
        def rsi(values, period=14):
            if len(values) < period+1:
                return None
            deltas = [values[i]-values[i-1] for i in range(1, len(values))]
            gains = [max(d,0) for d in deltas]
            losses = [max(-d,0) for d in deltas]
            avg_gain = sum(gains[:period]) / period
            avg_loss = sum(losses[:period]) / period
            for i in range(period, len(deltas)):
                avg_gain = (avg_gain*(period-1) + gains[i]) / period
                avg_loss = (avg_loss*(period-1) + losses[i]) / period
            if avg_loss == 0:
                return 100.0
            rs = avg_gain / avg_loss
            return 100 - (100 / (1+rs))
        r = rsi(closes)
        print(f"RSI14={r:.2f}" if r is not None else "RSI14=None")
        # SMA 50/200
        sma50 = sum(closes[-50:]) / 50 if len(closes) >= 50 else None
        sma200 = sum(closes[-200:]) / 200 if len(closes) >= 200 else None
        print(f"SMA50={sma50:.4f}" if sma50 else "SMA50=None", f"SMA200={sma200:.4f}" if sma200 else "SMA200=None")
        # golden cross check: did 50 cross above 200 recently?

            losses = [max(-d,0) for d in deltas]

            avg_loss = sum(losses[:period]) / period
            for i in range(period, len(deltas)):

                avg_loss = (avg_loss*(period-1) + losses[i]) / period
