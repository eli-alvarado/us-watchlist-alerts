import json, urllib.request, ssl
from datetime import datetime, timezone
from zoneinfo import ZoneInfo
tz = ZoneInfo("America/New_York")
ctx = ssl.create_default_context()
symbols = ["AAPL", "NVDA", "SPY"]
for sym in symbols:
    url = f"https://query1.finance.yahoo.com/v8/finance/chart/{sym}?interval=1m&range=1d&includePrePost=true"
    req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0 (compatible; watchlist/1.0)"})
    print("===", sym, "1m ===")
    try:
        with urllib.request.urlopen(req, context=ctx, timeout=20) as resp:
            print("STATUS", resp.status)
            data = json.loads(resp.read().decode("utf-8"))
            r = data["chart"]["result"][0]
            meta = r["meta"]
            keys = ["regularMarketPrice","regularMarketChangePercent","chartPreviousClose","previousClose","fulldayPrice","fulldayChange","fulldayChangePercent","regularMarketTime","postMarketPrice","postMarketChangePercent","currentTradingPeriod"]
            for k in keys:
                if k in meta:
                    print(f"  {k}: {meta[k]}")
            ts = r.get("timestamp") or []
            q = r["indicators"]["quote"][0]
            closes = q.get("close") or []
            # last non-null close
            last_i = None
            for i in range(len(closes)-1, -1, -1):
                if closes[i] is not None:
                    last_i = i
                    break
            if last_i is not None:
                dt = datetime.fromtimestamp(ts[last_i], tz=timezone.utc).astimezone(tz)
                print(f"  last 1m close: {closes[last_i]} at {dt.isoformat()}")
            print("  n timestamps", len(ts))
            print("  meta timezone", meta.get("timezone"), meta.get("exchangeTimezoneName"))
            # previousClose field dump of extra meta
            extra = {k: meta[k] for k in meta if "close" in k.lower() or "price" in k.lower() or "change" in k.lower() or k in ("symbol","regularMarketTime")}
            print("  extra", json.dumps(extra, default=str)[:2000])
    except Exception as e:
        print("ERROR", type(e).__name__, e)
